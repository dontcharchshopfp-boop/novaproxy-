Quick start
Подключение за минуту
example.js
const client = new OpenAI({
  baseURL: "https://novaproxy.onrender.com/v1",
  apiKey: "px_live_YOUR_KEY"
});

await client.chat.completions.create({
  model: "kimi-k3",
  messages: [{ role: "user", content: "Hello" }]
});
